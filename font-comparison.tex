\newcommand\Tall{\frac{\displaystyle\sum_{k=1}^n x_k}{1+p/q}}
\newcommand\Single{\begin{equation*}
 F=\left[\Tall+\frac{x^2+y^2}{z}+a+b\right]^2_j
\end{equation*}}
\newcommand\Broken{\begin{equation*}
 F&=\left[a+b+\\
 &\Tall+\\
 &(c+d)^2\right]^{n+1}_j
\end{equation*}}

$\displaystyle\frac{x}{1+\frac{p}{q}}\qquad\frac{x}{1+t^2}$\par\medskip
